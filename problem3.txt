        - Replace the top row in each adjacent face with the corresponding shifted row
    
    # Function to display a single face in a 2D format
    def display_face(face):
        - For each row in the face:
            - Print the elements of that row (based on 1D array indices)

    # Function to display the entire cube (all six faces)
    def display():
        - Print the front, back, left, right, up, and down faces
        - Call display_face for each face to show it in a 2D format
